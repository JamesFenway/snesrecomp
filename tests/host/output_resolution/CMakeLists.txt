cmake_minimum_required(VERSION 3.16)
project(snes_output_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
foreach(name output_policy output_window)
  add_executable(${name}_test ${name}_test.c)
  target_include_directories(${name}_test PRIVATE ../../../runner/src/desktop)
  if(MSVC)
    target_compile_options(${name}_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(${name}_test PRIVATE -UNDEBUG -Wall -Wextra)
    target_link_libraries(${name}_test PRIVATE m)
  endif()
  add_test(NAME ${name} COMMAND ${name}_test)
endforeach()

find_package(SDL3 CONFIG QUIET)
if(SDL3_FOUND)
  add_executable(output_sdl_test output_sdl_test.c)
  target_compile_definitions(output_sdl_test PRIVATE SNESRECOMP_SDL3=1 SDL_MAIN_HANDLED)
  target_include_directories(output_sdl_test PRIVATE ../../../runner/src/desktop)
  target_link_libraries(output_sdl_test PRIVATE SDL3::SDL3)
  if(MSVC)
    target_compile_options(output_sdl_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(output_sdl_test PRIVATE -UNDEBUG)
    target_link_libraries(output_sdl_test PRIVATE m)
  endif()
  add_test(NAME output_sdl_pixels COMMAND output_sdl_test)
  set_tests_properties(output_sdl_pixels PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=offscreen")
  add_executable(output_config_test ../config_roundtrip_test.c
      ../../../runner/src/desktop/mmx_config.c ../../../runner/src/util.c)
  target_compile_definitions(output_config_test PRIVATE SNESRECOMP_SDL3=1 SDL_MAIN_HANDLED)
  target_include_directories(output_config_test PRIVATE ../../../runner/src ../../../runner/src/desktop)
  target_link_libraries(output_config_test PRIVATE SDL3::SDL3)
  add_test(NAME output_config_roundtrip COMMAND output_config_test)
endif()
